     132  virtual function void calculate_and_store_addr(
     133      input bit only_aligned_addr

     135  //

     152  virtual function void parse_request;
     153    begin
     154      parsed_req.id = raw_req.id;
     155      parsed_req.start_address = raw_req.addr;
     156      parsed_req.size = raw_req.size;
     157      parsed_req.len = raw_req.len + 1;
     158      parsed_req.burst = raw_req.burst;
     159      //---------------------------------------------------
     160      // pre-calculate address details
     161      //---------------------------------------------------
     162      parsed_req.bytes_in_transfer = 2**raw_req.size;
     163      //incr addr
     164      parsed_req.total_bytes = parsed_req.bytes_in_transfer * parsed_req.len;    
     165      parsed_req.aligned_address = (parsed_req.bytes_in_transfer!=0) 
     166      ? (int'(parsed_req.start_address/parsed_req.bytes_in_transfer)) * parsed_req.bytes_in_transfer : 'hz;
     167      //wrap addr
     168      parsed_req.wrap_boundary = (int'(parsed_req.start_address/parsed_req.total_bytes)) * parsed_req.total_bytes;
     169      parsed_req.wrap_highest_address = parsed_req.wrap_boundary + parsed_req.total_bytes; 
     170    end
     171    //
     172  endfunction
     173  //
     174  //function to calculate transfer address
     175  //
     176  virtual function logic [31:0] calculate_next_addr(
     177      input int i,
     178      input bit only_aligned_addr
     179  );
     180  	//
     181  //	string header;
     182      logic [31:0] next_address;
     183  	//
     184      begin
     185  //	header = (parsed_req.wr_or_rd == 1) ? "WRITE" : "READ";
     186          //
     187          if(parsed_req.burst == 2'b00)begin  //fixed
     188              next_address = parsed_req.start_address;
     189          end
     190          else if (parsed_req.burst == 2'b01) begin //incr
     191              if(i==0) begin
     192                  next_address = parsed_req.start_address;
     193              end
     194              else 
     195                  next_address = parsed_req.aligned_address + i * parsed_req.bytes_in_transfer;
     196          end
     197          else if (parsed_req.burst == 2'b10) begin //wrap
     198              next_address = parsed_req.aligned_address + i * parsed_req.bytes_in_transfer;
     199              if(next_address == parsed_req.wrap_highest_address) begin
     200                  next_address = parsed_req.wrap_boundary;
     201              end
     202              else if(next_address > parsed_req.wrap_highest_address) begin
     203              //`uvm_info(get_type_name(), $sformatf("[%s][id: %0d][%0d]: next_wrap_address = 0x%08h", header, parsed_req.id[7:0], i, next_address),UVM_MEDIUM);
     204                  next_address = parsed_req.wrap_boundary + (next_address - parsed_req.wrap_highest_address); 
     205              end
     206              else begin
     207                  next_address = next_address;
     208              end   
     209          end //end of  "else if (m_mon_vif.awburst == 2'b10)""
     210          //recalculate address if needed 
     211          next_address = (only_aligned_addr) ? {next_address[31:2], 2'b00} : next_address;
     212          //
     213          return next_address;
     214      end
     215  endfunction

     226  virtual function void store_simulation_result(
     227      input resp_name actual_status
     228  );
     229      //
     230      result_info Transaction_result;
     231      result_q tmp_q; 
     232      //
     233      begin
     234  	Transaction_result.wr_or_rd = raw_req.wr_or_rd;
     235          Transaction_result.id = raw_req.id;
     236          Transaction_result.address = raw_req.addr;
     237          Transaction_result.len = raw_req.len;

     239          Transaction_result.burst = raw_req.burst;
     240          Transaction_result.resp = resp_name'(actual_status);
     241          Transaction_result.case_matches = match;
     242          Transaction_result.case_mismatches = mismatch;
     243  	//
     244  	tmp_q = {};
     245          //
     246  	if(sim_result_arr.exists(raw_req.id[7:0])) begin
     247  		tmp_q = sim_result_arr[raw_req.id[7:0]];
     248  		tmp_q.push_back(Transaction_result);
     249  		sim_result_arr[raw_req.id[7:0]] = tmp_q;
     250  		`uvm_info(get_type_name(), 
     251  		$sformatf("[EXISTING]Store SIM RESULT of [%s][%0d(ID)]",
     252  	       	raw_req.wr_or_rd ? "WRITE" : "READ_", raw_req.id[7:0]), UVM_LOW)
     253  	end
     254  	else begin
     255  		tmp_q.push_back(Transaction_result);
     256  		sim_result_arr[raw_req.id[7:0]] = tmp_q;
     257  		`uvm_info(get_type_name(), 
     258  		$sformatf("[NEW]Store SIM RESULT of [%s][%0d(ID)]", 
     259  		raw_req.wr_or_rd ? "WRITE" : "READ_", raw_req.id[7:0]), UVM_LOW)
     260  	end
     261      end

     365  //
     366  virtual function void display_rsp_report();
     367      resp_name expected_bresp, actual_bresp;
     368      resp_q expected_tmp_q, actual_tmp_q;
     369      int pass_cnt = 0;
     370      int fail_cnt = 0;
     371      id_of_burst id;
     372       `ifdef PRINT_TO_SUM_FILE
     373  	$fdisplay(sim_summary_file,"================== BCHANNEL REPORT =================");
     374      `else 
     375      `uvm_info(get_type_name(),"================== BCHANNEL REPORT =================", UVM_LOW);
     376      `endif 
     377  
     378      if(expected_bresp_array.num() > 0 && actual_bresp_array.num() > 0) begin
     379  	id = 0;
     380  	for(int i = 0; i < 256; i++) begin
     381  	if(expected_bresp_array.exists(id)) begin //existence
     382  		if(actual_bresp_array.exists(id)) begin
     383  			actual_tmp_q = {};
     384  			actual_tmp_q = actual_bresp_array[id];
     385  			expected_tmp_q = {};
     386  			expected_tmp_q = expected_bresp_array[id];
     387  			//do compare
     388  			foreach(actual_tmp_q[i]) begin
     389  				actual_bresp = actual_tmp_q[i]; 
     390  				expected_bresp = expected_tmp_q[i]; 
     391  			`ifdef PRINT_TO_SUM_FILE
     392  				$fdisplay(sim_summary_file,
     393  				$sformatf("[COMPARE_BCHANNEL]: actual_bid=%0d---expected_bid=%0d", id, id));
     394  				//
     395  				if(actual_bresp == expected_bresp) begin
     396  				$fdisplay(sim_summary_file,
     397  				$sformatf("[----------------][PASS]: actual_bresp=%s___expected_bresp=%s",
     398  			       	actual_bresp.name(), expected_bresp.name())); 
     399  				pass_cnt++;
     400  				end
     401  				else begin
     402  				$fdisplay(sim_summary_file,
     403  				$sformatf("[----------------][FAIL]: actual_bresp=%s___expected_bresp=%s",
     404  			       	actual_bresp.name(), expected_bresp.name())); 
     405  				fail_cnt++;
     406  				end
     407  		    	`else 
     408  				`uvm_info(get_type_name(),
     409  				$sformatf("[COMPARE_BCHANNEL]: actual_bid=%0d---expected_bid=%0d", id, id), UVM_LOW);
     410  				//
     411  				if(actual_bresp == expected_bresp) begin
     412  				`uvm_info(get_type_name(),
     413  				$sformatf("[----------------][PASS]: actual_bresp=%s___expected_bresp=%s",
     414  			       	actual_bresp.name(), expected_bresp.name()), UVM_LOW); 
     415  				pass_cnt++;
     416  				end
     417  				else begin
     418  				`uvm_info(get_type_name(),
     419  				$sformatf("[----------------][FAIL]: actual_bresp=%s___expected_bresp=%s",
     420  			       	actual_bresp.name(), expected_bresp.name()), UVM_LOW); 
     421  				fail_cnt++;
     422  				end
     423  			`endif
     424  				actual_bresp = resp_name'(2'b00);
     425  				expected_bresp = resp_name'(2'b00);
     426  			end
     427  		end
     428  		else begin
     429  		    `ifdef PRINT_TO_SUM_FILE
     430  			$fdisplay(sim_summary_file,$sformatf("id = %0d does not appear in actual array", id));
     431  		    `else 
     432  		    `uvm_info(get_type_name(),$sformatf("id = %0d does not appear in actual array", id), UVM_LOW);
     433  		    `endif 
     434  		end
     435  	end//end of existence
     436  	id++;
     437  	end//end of for	
     438      end
     439      else begin
     440  	    if(expected_bresp_array.num() == 0) begin
     441  		    `uvm_info(get_type_name(),"EMPTY expected_bresp_array", UVM_LOW);
     442  	    end
     443  	    if(actual_bresp_array.num() == 0) begin
     444  		    `uvm_info(get_type_name(),"EMPTY actual_bresp_array", UVM_LOW);
     445  	    end
     446      end
     447      //
     448      `ifdef PRINT_TO_SUM_FILE
     449      $fdisplay(sim_summary_file, $sformatf("WRITE_RSP CHECK: %0d(PASS)___%0d(FAIL)", pass_cnt, fail_cnt))
     450      $fdisplay(sim_summary_file,"=======================================================");
     451  	$fclose(sim_summary_file);
     452      `else
     453      `uvm_info(get_type_name(), $sformatf("WRITE_RSP CHECK: %0d(PASS)___%0d(FAIL)", pass_cnt, fail_cnt), UVM_LOW)
     454      `uvm_info(get_type_name(),"=======================================================", UVM_LOW);
     455      `endif 
     456  endfunction

     523  	init_slave_addr(APB_BASE_END_ADDR_QUEUE_WIDTH, 32'h0, 32'h0000_0FFF);
     524  endfunction
     525  //
     526  virtual task run_phase(uvm_phase phase);
     527  	super.run_phase(phase);
     528  	//
     529  	forever begin
     530  	    wait_apb_transfer(3);
     531  	    #(`CLK_CYCLE);
     532  	    if(compare_start) begin
     533  		    if(apb_content_q[0].pwrite) begin

     539  			`uvm_info(get_type_name(), $sformatf("[READ] COMPARE TRANSACTION START"), UVM_MEDIUM);
     540  			compare_transaction(1'b0);
     541  		    end
     542  	    end
     543  	end
     544  endtask

     722      //
     723  task scoreboard::wait_apb_transfer(input int timeout_consecutive_times);
     724  //
     725  compare_start = 0;
     726  wait(apb_rst_flag == 0);
     727  fork
     728  	begin: thread_1
     729  		@(posedge apb_rst_flag);
     730  		   `uvm_info(get_type_name(), $sformatf("[%0t (ns)] apb_rst_flag posedge", $time), UVM_MEDIUM)
     731  	end
     732  	//
     733  	begin: thread_2
     734  		wait(apb_content_q.size() > 0); 
     735  		compare_start = 1'b1;
     736  		clr_member();
     737  	end
     738  	//
     739  	begin: thread_3
     740  		#APB_TIMEOUT;
     741  		sim_counter++;
     742  		if(sim_counter == timeout_consecutive_times) done = 1;
     743  		//
     744  		if(~done) begin
     745  		`uvm_warning(get_type_name(), $sformatf("at[%0t (ns)]APB TRANSFER TIMEOUT!!!", $time));
     746  		end
     747  	end
     748  join_any
     749  //
     750  disable fork;
     751  endtask
     752  //
     753  //
     754  //
     755  task scoreboard::compare_transaction(
     756  	 input bit wr_en
     757  );
     758  	wait(axi_rst_flg == 1'b0);
     759          fork 
     760              begin //FIRST THREAD
     761                  @(posedge axi_rst_flg);
     762                  `uvm_info(get_type_name(), $sformatf("RISING EDGE AXI_RST_FLAG!!!"), UVM_MEDIUM);
     763              end
     764              begin //SECOND_THREAD
     765  		init_attribute();
     766  		Fetch_Valid_Req(wr_en);
     767  		Do_Comparison(wr_en);
     768              end //end of SECOND_THREAD
     769          join_any
     770          disable fork;
     771  endtask

     775  //Operation Order:
     776  //--: check unvalid req -> pop unvalid data
     777  //--: return valid req and do compare     
     778  task scoreboard:: Fetch_Valid_Req(input bit wr_en);
     779      //
     780      bit queue_not_empty;
     781      bit error_case;
     782      //
     783      begin//begin_end block
     784          while (1) begin
     785              queue_not_empty = (wr_en) ? (wr_req_q.size() > 0) : (rd_req_q.size() > 0);
     786              if(queue_not_empty) begin
     787  		raw_req = axi_req_item::type_id::create("raw_req");
     788                  raw_req = (wr_en) ? (wr_req_q.pop_front()) : (rd_req_q.pop_front());
     789  		error_case = (raw_req.size != 3'b010);//unsupported transactions
     790                  //
     791                  if(!Check_Valid_Addr(raw_req.addr) || (error_case)) begin
     792  		    
     793                      if(wr_en) begin
     794  			if(error_case) 
     795                          Store_Expected_Bresp(raw_req.id[7:0], resp_name'(2'b10));//pslverr
     796  			else 
     797                          Store_Expected_Bresp(raw_req.id[7:0], resp_name'(2'b11)); //decerr
     798                      end
     799                      Pop_unvalid_data(wr_en);
     800                  end
     801                  else begin
     802  		    break;
     803                  end
     804              end //end of if queue_not_empty
     805              else begin
     806  	    `uvm_fatal(get_type_name(), $sformatf("[FETCH_VALID_REQ]: UNAVAILABLE %s REQ", wr_en ? "READ" : "WRITE"))
     807              end
     808      end//end of while(1)
     809      end//end of begin_end block
     810  endtask

     812      function bit scoreboard::Check_Valid_Addr(input logic [31:0] start_address);
     813      	//
     814  	bit flag;
     815          begin
     816              flag = (start_address <= apb_base_end_addr_q[`SLAVE_CNT*2-1]);
     817          end
     818  	//
     819  	return flag;
     820      endfunction
     821      //
     822      task scoreboard::Pop_unvalid_data(input bit wr_en);
     823          //
     824  	bit data_available = 0;
     825  	bit exit_data_while_loop = 0;
     826          data_q axi_rdata_tmp_q = {};
     827  	//
     828          begin//START
     829              //
     830              while(1) begin
     831  		    data_available = (wr_en) ? (axi_wdata_q.size() > 0) : (axi_rdata_array[raw_req.id[7:0]].size() > 0);
     832                  if (data_available) begin
     833  			// clr_content();
     834  		   axi_content = axi_data_item::type_id::create("axi_content");
     835  		   if(wr_en) begin
     836  			axi_content = axi_wdata_q.pop_front();
     837  		    `uvm_info(get_type_name(), $sformatf("[INVALID_WDATA]: REMOVED FROM QUEUE!!!"), UVM_MEDIUM)
     838  		    end
     839  		    else begin
     840  			axi_rdata_tmp_q = axi_rdata_array[raw_req.id[7:0]];
     841  			axi_content = axi_rdata_tmp_q.pop_front();
     842  			axi_rdata_array[raw_req.id[7:0]] = axi_rdata_tmp_q;
     843  		    `uvm_info(get_type_name(), $sformatf("[INVALID_RDATA]: REMOVED FROM QUEUE!!!"), UVM_MEDIUM)
     844  		    end
     845  		    //axi_content.print();
     846  		    last_beat = axi_content.last;
     847  		    axi_data_wait_cnt = 0;
     848  		    //
     849  		    if(last_beat == 1'b1) begin
     850  			    `uvm_info(get_type_name(), $sformatf("[INVALID LAST_DATA]: REMOVED FROM QUEUE!!!"), UVM_MEDIUM)
     851  			handle_last_transfer(axi_content.resp);
     852  			break; //exit while(1) 
     853  		    end
     854                  end
     855  	       	else begin
     856                      axi_data_wait_cnt++;
     857                      #(`CLK_CYCLE);
     858      //wr_en
     859      //axi_or_apb;
     860      //output bit break_enable;
     861                      handle_wait_data(wr_en, 1'b1, exit_data_while_loop);
     862                      if(exit_data_while_loop==1) begin
     863                          #(`CLK_CYCLE);
     864                          break;//exit while(1) 
     865                      end
     866                  end
     867  	end //end of while(1)
     868  	end //end of START
     869      endtask
     870  //
     871  //
     872  //
     873  task scoreboard::Do_Comparison(input bit wr_en);
     874          //
     875  	int transfer_index = 0;
     876  	resp_name bresp = resp_name'(2'b00);
     877  	data_q axi_rdata_tmp_q = {};
     878          //
     879  	begin //begin_end block
     880              //at first
     881  	    calculate_and_store_addr(1'b1);
     882  	    assert(axi_addr_q.size() > 0);
     883              //**************************************************************
     884              //--------------------AXI - APB
     885              //**************************************************************
     886  	while(1) begin
     887  		    `uvm_info(get_type_name(), $sformatf("[DO_COMPARE]: TRANSFER WAIT!!!"), UVM_MEDIUM)
     888  		    wait(apb_content_q.size() > 0);
     889  		    if(wr_en) begin
     890  			    wait(axi_wdata_q.size() > 0);
     891  		    end
     892  		    else begin
     893  			    wait(axi_rdata_array.exists(raw_req.id[7:0]) && axi_rdata_array[raw_req.id[7:0]].size() > 0);
     894  		    end
     895  		    `uvm_info(get_type_name(), $sformatf("[DO_COMPARE]: TRANSFER START!!!"), UVM_MEDIUM)
     896  		//clr_content();
     897  		//----APB
     898  		   apb_content = apb_item::type_id::create("apb_content");
     899  		   apb_content = apb_content_q.pop_front();
     900  		//----AXI
     901  		   axi_content = axi_data_item::type_id::create("axi_content");
     902  		    if(wr_en) begin
     903  			axi_content = axi_wdata_q.pop_front();
     904  		    `uvm_info(get_type_name(), $sformatf("[DO_COMPARE]: READY FOR WR_TRANSFER COMPARE!!!"), UVM_MEDIUM)
     905  		    end
     906  		    else begin
     907  			axi_rdata_tmp_q = axi_rdata_array[raw_req.id[7:0]];
     908  			axi_content = axi_rdata_tmp_q.pop_front();
     909  			axi_rdata_array[raw_req.id[7:0]] = axi_rdata_tmp_q;
     910  		    `uvm_info(get_type_name(), $sformatf("[DO_COMPARE]: READY FOR RD_TRANSFER COMPARE!!!"), UVM_MEDIUM)
     911  		    end
     912  		//----COMPARE
     913  		convert_axi_to_compare(axi_content);
     914  		convert_apb_to_compare(apb_content);
     915  		    compare_transfer(transfer_index);
     916  		    transfer_index++;
     917  		    `uvm_info(get_type_name(), $sformatf("[DO_COMPARE]: TRANSFER DONE!!!"), UVM_MEDIUM)
     918  		    //
     919  		    //store resp
     920  		    bresp = resp_name'(bresp | apb_transfer.resp);
     921  		    last_beat = axi_content.last;
     922  		    //
     923  		    if(last_beat == 1'b1) begin
     924  			    if(wr_en) begin
     925  			    Store_Expected_Bresp(raw_req.id[7:0], bresp);
     926  			    end
     927  		    `uvm_info(get_type_name(),
     928  		     $sformatf("[DO_COMPARE]: %s TRANSACTION DONE", wr_en ? "WRITE" : "READ"), UVM_MEDIUM)
     929  		    	//
     930  			handle_last_transfer(axi_content.resp);
     931  			break; //exit while(1) 
     932  		    end
     933  	    end//end of while(1)
     934  	end//end of begin_end block
     935      endtask

     960                  //`else

     978              end
     979          //end
     980          //
     981          `ifdef PRINT_TO_COMPARE_FILE

     983          `endif
     984          //
     985      end
     986  endtask

     988  //
     989  //
     990  task scoreboard::handle_wait_data(input bit write_enable, input bit axi_or_apb, output bit break_enable);
     991      //
     992      resp_name wait_resp;
     993      string header;
     994      //
     995      begin
     996          //
     997          if (axi_data_wait_cnt > TIME_OUT_BOUNDARY) begin
     998              wait_resp = resp_name'(2'b01);
     999              store_simulation_result(wait_resp);
     1000              //trans_completed = 1'b1;
     1001              if(axi_or_apb == 1) begin
     1002                  if((write_enable==1'b0) && (axi_rdata_array.size() == 0)) begin
     1003                      `uvm_info(get_type_name(),
     1004                      $sformatf("TRANS[READ][%0d]:Timeout Timeout Timeout--axi_rdata_array is EMPTY", raw_req.id[7:0]), UVM_MEDIUM);
     1005                  end
     1006                  //
     1007                  if((write_enable==1'b1) && (axi_wdata_q.size() == 0)) begin
     1008                      `uvm_info(get_type_name(),
