     3  `define PSEL_REGISTER
     4  //
     5  module apb_master(
     6  	pclk,
     7  	preset_n,
     8  	//
     9  	//control signal
     10  	//
     11  	transfer_i,
     12  	enable_i,
     13  	nonexist_transfer_i,
     14  	addr_i,
     15  	prot_i,
     16  	grant_to_write_i,
     17  	wstrb_to_apb_i,	
     18  	wdata_to_apb_i,	
     19  	//APB interface
     20  	paddr,
     21  	pprot,
     22  	psel,
     23  	penable,
     24  	pwrite,
     25  	pwdata,
     26  	pstrb,
     27  	pready,
     28  	prdata,
     29  	pslverr,
     30  //send to AXI transaction controller
     31  	prdataX_o,
     32  	pslverrX_o,
     33  	master_ctrl_o
     34  	//
     35  	//
     36  );
     37  //*********************************************************
     38  //PORTS
     39  //*********************************************************
     40  input logic                    pclk;
     41  input logic                    preset_n;
     42  //
     43  input logic 			transfer_i;
     44  input logic [`SLAVE_CNT-1:0]    enable_i;
     45  input logic 			nonexist_transfer_i;
     46  input logic [31:0]		addr_i;
     47  input logic [2:0]		prot_i;
     48  input logic 			grant_to_write_i;
     49  input logic [3:0]		wstrb_to_apb_i;
     50  input logic [31:0]		wdata_to_apb_i;	
     51  //
     52  output logic [31:0]             paddr;
     53  output logic [2:0]              pprot;
     54  output logic [`SLAVE_CNT-1:0]   psel;
     55  output logic                    penable;
     56  output logic                    pwrite;
     57  output logic [31:0]             pwdata;
     58  output logic [3:0]              pstrb;
     59  input  logic [`SLAVE_CNT-1:0][31:0] prdata;
     60  input  logic [`SLAVE_CNT-1:0]   pready;
     61  input  logic [`SLAVE_CNT-1:0]   pslverr;
     62  //send to AXI transaction controller
     63  output logic [31:0] prdataX_o;
     64  output logic pslverrX_o;
     65  output logic [4:0] master_ctrl_o;
     66  //*********************************************************
     67  //INTERNAL VARIABLES
     68  //*********************************************************
     69  //APB state machine
     70  typedef enum logic [1:0] {P_IDLE, SETUP, ACCESS} apb_st;
     71  apb_st apb_cs, apb_ns;
     72  //
     73  logic [`SLAVE_CNT-1:0] out_pready;
     74  logic [`SLAVE_CNT-1:0] out_pslverr;
     75  logic [`SLAVE_CNT-1:0][31:0] out_prdata;
     76  logic [`SLAVE_CNT-1:0] preadyX;
     77  logic invalid_psel;
     78  logic store_rdata, fetch_wdata, latch_resp, set_up_phase, addr_incr_en;
     79  //
     80  assign master_ctrl_o = {store_rdata, fetch_wdata, latch_resp, set_up_phase, addr_incr_en};
     81  //
     82    `ifdef PSEL_REGISTER
     83  	  logic [`SLAVE_CNT-1:0] true_psel_reg;
     84  	  logic false_psel_reg;
     85  	  //
     86  	  always_ff@(posedge pclk, negedge preset_n) begin
     87  		  if(~preset_n) begin
     88  			  true_psel_reg <= {`SLAVE_CNT{1'b0}};
     89  			  false_psel_reg <= 1'b0;
     90  		  end
     91  		  else begin
     92  			  true_psel_reg <= enable_i;
     93  			  false_psel_reg <= (|enable_i) ? 1'b0 : 1'b1;
     94  		  end
     95  	  end
     96    `endif
     97    //
     98    //APB state regs
     99    //
     100    always_ff @(posedge pclk, negedge preset_n) begin
     101      if(~preset_n)
     102  	  apb_cs[1:0] <= P_IDLE;
     103  	else
     104  	  apb_cs[1:0] <= apb_ns[1:0];
     105    end
     106  //
     107  //
     108  //
     109    always_comb begin
     110  	apb_ns = apb_cs;
     111  	pprot = 0;
     112  	psel = 0;
     113  	invalid_psel = 0;
     114  	penable = 0;
     115  	paddr = 0;
     116  	pwrite = 0;
     117  	pwdata = 0;
     118  	pstrb = 0;
     119  	store_rdata = 0;
     120  	fetch_wdata = 0;
     121  	latch_resp = 0;
     122  	set_up_phase = 0;
     123  	addr_incr_en = 0;
     124  	//
     125      case(apb_cs[1:0])
     126  		P_IDLE: begin
     127  			if(transfer_i) begin
     128  				apb_ns[1:0] = SETUP;
     129  			end
     130  			else begin
     131  				apb_ns[1:0] = P_IDLE;
     132  			end
     133  		end
     134  		SETUP: begin
     135  			apb_ns[1:0] = ACCESS;
     136  			`ifdef PSEL_REGISTER
     137  			psel = true_psel_reg;
     138  			invalid_psel = false_psel_reg;
     139  			`elsif
     140  			psel = enable_i;
     141  			invalid_psel = (|enable_i) ? 1'b0 : 1'b1;
     142  			`endif
     143  			penable = 1'b0;
     144  			pprot[2:0] = prot_i;
     145  			paddr = addr_i;
     146  			pwrite = (grant_to_write_i) ? 1'b1: 1'b0;
     147  			pwdata = (grant_to_write_i) ? wdata_to_apb_i[31:0] : 32'd0;
     148  		    pstrb[3:0] = (grant_to_write_i) ? wstrb_to_apb_i[3:0] : 4'h0;
     149  		    //
     150  			store_rdata = 0;
     151  			fetch_wdata = 0;
     152  			latch_resp = 0;
     153  		        set_up_phase = 1'b1;
     154  			addr_incr_en = 0;
     155  		end
     156  		ACCESS: begin
     157  			if(preadyX) begin
     158  				//
     159  				latch_resp = (grant_to_write_i) ? 1'b1 : 1'b0;
     160  				addr_incr_en = 1'b1;
     161  				store_rdata = (grant_to_write_i) ? 1'b0 : 1'b1;
     162  				fetch_wdata = (grant_to_write_i) ? 1'b1 : 1'b0;
     163  				//
     164  				if(transfer_i) begin
     165  					apb_ns[1:0] = SETUP;
     166  				end
     167  				else begin
     168  					apb_ns[1:0] = P_IDLE;
     169  				end
     170  			end //end of preadyX
     171  			else begin
     172  				apb_ns[1:0] = ACCESS;
     173  			end
     174  			`ifdef PSEL_REGISTER
     175  			psel = true_psel_reg;
     176  			invalid_psel = false_psel_reg;
     177  			`elsif
     178  			psel = enable_i;
     179  			invalid_psel = (|enable_i) ? 1'b0 : 1'b1;
     180  			`endif
     181  			penable = 1'b1;
     182  			pprot[2:0] = prot_i;
     183  			paddr = addr_i;
     184  			pwrite = (grant_to_write_i) ? 1'b1: 1'b0;
     185  			pwdata = (grant_to_write_i) ? wdata_to_apb_i[31:0] : 32'd0;
     186  		    pstrb[3:0] = (grant_to_write_i) ? wstrb_to_apb_i[3:0] : 4'h0;
     187  			//
     188  		end
     189  		default: begin
     190  			apb_ns[1:0] = P_IDLE;
     191  			pprot = 0;
     192  			psel = 0;
     193  			invalid_psel = 0;
     194  			penable = 0;
     195  			paddr = 0;
     196  			pwrite = 0;
     197  			pwdata = 0;
     198  			pstrb = 0;
     199  			store_rdata = 0;
     200  			fetch_wdata = 0;
     201  			latch_resp = 0;
     202  			set_up_phase = 0;
     203  			addr_incr_en = 0;
     204  		end
     205  	endcase
     206  end
     207  //pslverrX, preadyX
     208    assign preadyX  = |out_pready[`SLAVE_CNT-1:0] | invalid_psel | nonexist_transfer_i; 
     209    assign pslverrX_o = |out_pslverr[`SLAVE_CNT-1:0]| invalid_psel | nonexist_transfer_i; 
     210    generate
     211      genvar i;
     212  	for (i = 0; i <= `SLAVE_CNT-1; i = i + 1) begin: decPreadyAndPslverr
     213  	  assign out_pready[i]  = psel[i] & pready[i];
     214  	  assign out_pslverr[i] = psel[i] & pslverr[i];
     215  	end
     216    endgenerate
     217    //prdataX
     218    assign prdataX_o = out_prdata[`SLAVE_CNT-1];
     219    assign out_prdata[0] = psel[0] ? prdata[0] : 32'd0;
     220    //
     221    generate
     222      genvar j;
     223  	for(j = 1; j <= `SLAVE_CNT-1; j = j + 1) begin: decPrdata
     224  	  assign out_prdata[j] = psel[j] ? prdata[j] : out_prdata[j-1];
     225  	end
     226    endgenerate
     227  //
     228  //
     229  endmodule
