     15  //this class takes responsibility for generating stimulus
     16  typedef axi_transaction#(DW,AW1) axi_item;
     17  //
     18  class stimulus_generator extends uvm_sequence#(axi_item); 
     19  	`uvm_object_utils(stimulus_generator)
     20  	//
     21  	mailbox #(axi_item) mb;
     22  	axi_item trans_h;
     23  	int total_no_test;
     24          bit reconfigure_id;
     25  	//
     26  	function new(string name = "stimulus_generator");
     27  		super.new(name);
     28  	endfunction
     29  	//
     30  	virtual function void gen_item();
     31  		mb = new(total_no_test);
     32  		//mb = new();
     33  		repeat(total_no_test) begin
     34  			trans_h = axi_item::type_id::create("axi_item");
     35  			assert(trans_h.randomize())
     36  			else `uvm_error(get_type_name(), "randomize axi_item FAILED")
     37  			if(!mb.try_put(trans_h)) `uvm_error(get_type_name, "try_put item into mb FAIL!!!");
     38  		end
     39  	endfunction
     40  	//
     41  endclass

     49  class AxiMasterWriteSeq #(DW, AW1) extends stimulus_generator;
     50      //register UVM factory
     51      `uvm_object_param_utils(AxiMasterWriteSeq#(DW, AW1))
     52      int no_test;
     53      // Constructor
     54      function new (string name = "AxiMasterWriteSeq");
     55          super.new(name);
     56      endfunction

     58      extern virtual task body();
     59

     62  //main tasks in sequence
     63      
     64  //-- body()
     65  //
     66  task AxiMasterWriteSeq::body();
     67  `uvm_info(get_type_name(), $sformatf("WRITE no_test = %0d", no_test), UVM_LOW);
     68      repeat(no_test) begin
     69        // wait_for_grant();
     70        //send_request(trans_h); //must pair wait_for_grant()
     71                trans_h = axi_item::type_id::create("axi_w_trans");
     72  
     73         if(mb.try_get(trans_h)) begin
     74  	      //send item to sequencer
     75  	      start_item(trans_h); //block until get_next_item() from driver
     76  	      trans_h.set_bytes_in_beat(3'b010);
     77  	      finish_item(trans_h);
     78        end
     79        else begin
     80  		`uvm_fatal(get_type_name(), $sformatf("wr_stimulus not available!!!"))
     81        end
     82      end
     83  endtask

     85  //--------------AXI MASTER READ-------------------------
     86  //
     87  class AxiMasterReadSeq#(DW,AW1) extends stimulus_generator;
     88    `uvm_object_param_utils(AxiMasterReadSeq#(DW,AW1))
     89    //
     90      logic [7:0] r_id;
     91      int no_test;
     92      //
     93      function new(string name = "read transaction");
     94          super.new(name);
     95      endfunction 
     96      //
     97      extern virtual task body();
     98  endclass
     99  //
     100  //
     101  task AxiMasterReadSeq::body();
     102  	r_id = 0;
     103  	`uvm_info(get_type_name(), $sformatf("RD no_test = %0d", no_test), UVM_LOW);
     104  	repeat(no_test) begin
     105  	      //r_mb.get(tmp);
     106  	      //trans_h.copy(tmp); 
     107  	      //start_item does not recognize mailbox's element; therefore, we use copy() method to create new object
     108  	      //
     109  	      trans_h = axi_item::type_id::create("axi_r_trans");
     110  	      if(mb.try_get(trans_h)) begin
     111  		      //send item to sequencer
     112  			`uvm_info(get_type_name(), $sformatf("[START]trans_h!!!"), UVM_HIGH);
     113  		      start_item(trans_h);
     114  		      trans_h.set_bytes_in_beat(3'b010);
     115  		      if(reconfigure_id) begin
     116  			      trans_h.set_id(r_id);
     117  		      end
     118  		      trans_h.set_aligned_addr();
     119  		      //item sent to driver done
     120  		      finish_item(trans_h);
     121  			`uvm_info(get_type_name(), $sformatf("[DONE]trans_h!!!"), UVM_HIGH);
     122  		      r_id++;
     123        	      end
     124  	      else begin
     125  		`uvm_fatal(get_type_name(), $sformatf("rd_stimulus unavailable!!!"))
     126  	      end
     127  	end
     128  endtask

     130  //Test_Response Sequence
     131  //
     132  class rsp_seq extends stimulus_generator;
     133  	`uvm_object_utils(rsp_seq)
     134  	//
     135  	int no_test_slverr1;	
     136  	int no_test_slverr2;	
     137  	int no_test_decerr;	
     138  	//
     139  	function new (string name = "response seq");
     140  		super.new(name);
     141  	endfunction
     142  	//
     143   	function void gen_item();
     144  		mb = new();
     145  		//disallowed size
     146  		repeat(no_test_slverr1) begin
     147  			trans_h = axi_item::type_id::create("axi_item");
     148  			assert(trans_h.randomize() with {
     149  				trans_h.size != 3'b010;
     150  				trans_h.is_valid == 1'b1;
     151  				})
     152  			else `uvm_error(get_type_name(), "randomize axi_item FAILED")
     153  			//mb.put(trans_h);
     154  			if(!mb.try_put(trans_h)) `uvm_error(get_type_name, "try_put item into mb FAIL!!!");
     155  		end
     156  		//unaligned addr for WRAP, FIXED 
     157  		repeat(no_test_slverr2) begin
     158  			trans_h = axi_item::type_id::create("axi_item");
     159  			assert(trans_h.randomize() with {
     160  				trans_h.size == 3'b010;
     161  				trans_h.is_valid == 1'b1;
     162  				trans_h.burst != INCR;
     163  				trans_h.addr[1:0] != 2'b00;
     164  				})
     165  			else `uvm_error(get_type_name(), "randomize axi_item FAILED")
     166  			//mb.put(trans_h);
     167  			if(!mb.try_put(trans_h)) `uvm_error(get_type_name, "try_put item into mb FAIL!!!");
     168  		end
     169  
     170  		//
     171  		repeat(no_test_decerr) begin
     172  			trans_h = axi_item::type_id::create("axi_item");
     173  			assert(trans_h.randomize() with {
     174  				trans_h.size == 3'b010;
     175  				trans_h.is_valid == 1'b0;
     176  				})
     177  			else `uvm_error(get_type_name(), "randomize axi_item FAILED")
     178  			//mb.put(trans_h);
     179  			if(!mb.try_put(trans_h)) `uvm_error(get_type_name, "try_put item into mb FAIL!!!");
     180  		end
     181  	endfunction
     182  	//
     183  	virtual task body();
     184  		repeat(no_test_slverr1 + no_test_slverr2 + no_test_decerr) begin
     185  		      trans_h = axi_item::type_id::create("axi_trans");
     186  		      if(mb.try_get(trans_h)) begin
     187  			      //send item to sequencer
     188  				`uvm_info(get_type_name(), $sformatf("[START]trans_h!!!"), UVM_HIGH);
     189  			      start_item(trans_h);
     190  			      trans_h.set_aligned_addr();
     191  			      //item sent to driver done
     192  			      finish_item(trans_h);
     193  				`uvm_info(get_type_name(), $sformatf("[DONE]trans_h!!!"), UVM_HIGH);
     194  			end
     195  			else begin
     196  				`uvm_fatal(get_type_name(), "transaction is UNAVAILABLE")
     197  			end
     198        		end
     199  	endtask
     200  endclass
