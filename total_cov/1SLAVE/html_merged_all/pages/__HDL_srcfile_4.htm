1  //===================================================================================
     2  // File name: x2p_core.sv
     3  // Project  : X2P
     4  // Function : IP core design of AXI to APB bridge
     5  //===================================================================================
     6  module axi_transaction_controller (// AXI protocol
     7      aclk,
     8      aresetn,
     9  // Address write channel
     10  	awvalid,
     11  	awready,
     12  	awaddr,
     13  	awsize,
     14  	awlen,
     15  	awburst,
     16  	awid,
     17  	awprot,
     18  	// Address read channel
     19  	arvalid,
     20  	arready,
     21  	araddr,
     22  	arsize,
     23  	arlen,
     24  	arburst,
     25  	arid,
     26  	arprot,
     27  	// Write data channel
     28  	wvalid,
     29  	wready,
     30  	wdata,
     31  	wstrb,
     32  	wlast,
     33  	// Read data channel
     34  	rvalid,
     35  	rready,
     36  	rlast,
     37  	rresp,
     38  	rid,
     39  	rdata,
     40  	// Write respond channel
     41  	bvalid,
     42  	bready,
     43  	bresp,
     44  	bid,
     45  	//arbiter input
     46  	rd_trans_done_i,
     47  	wr_trans_done_i,
     48  	//apb master input
     49  	prdata_i,
     50  	err_of_transfer_i,
     51  	write_to_rd_sfifo_i,
     52  	read_from_wd_sfifo_i,
     53  	dec_error_i,
     54  	latch_resp_i,
     55  	//req
     56  	sfifo_aw_empty_o,
     57  	sfifo_ar_empty_o,
     58  	//data
     59  	sfifo_wd_almost_empty_o,
     60  	sfifo_rd_almost_full_o,
     61  	//
     62  	  write_burst_addr_o,
     63  	  write_burst_len_o,
     64  	  write_burst_size_o,
     65  	  write_burst_name_o,
     66  	  write_burst_prot_o,
     67  		//	
     68  	  read_burst_addr_o,
     69  	  read_burst_len_o,
     70  	  read_burst_size_o,
     71  	  read_burst_name_o,
     72  	  read_burst_prot_o,
     73  	  //
     74  	  wdata_to_apb_o,
     75  	  wstrb_to_apb_o
     76  	  //
     77  );
     78    //iclude parameter file
     79    import parameter_pkg::*; 
     80    //******************************************************
     81    //ports declaration
     82    //******************************************************
     83    input logic                     aclk;
     84    input logic                     aresetn;
     85    // Address write channel
     86    input  logic                    awvalid;
     87    input  logic [31:0]             awaddr;
     88    input  logic [2:0]              awsize;
     89    input  logic [7:0]              awlen;
     90    input  logic [1:0]              awburst;
     91    input  logic [7:0]              awid;
     92    input  logic [2:0]              awprot;
     93    output logic                    awready;
     94    // address read channel
     95    input  logic                    arvalid;
     96    input  logic [31:0]             araddr;
     97    input  logic [2:0]              arsize;
     98    input  logic [7:0]              arlen;
     99    input  logic [1:0]              arburst;
     100    input  logic [7:0]              arid;
     101    input  logic [2:0]              arprot;
     102    output logic                    arready;
     103    //write data channel
     104    input  logic                    wvalid;
     105    input  logic [31:0]             wdata;
     106    input  logic [3:0]              wstrb;
     107    input  logic                    wlast;
     108    output logic                    wready;
     109    //read data channel
     110    input  logic                    rready;
     111    output logic                    rvalid;
     112    output logic [1:0]              rresp;
     113    output logic                    rlast;
     114    output logic [7:0]              rid;
     115    output logic [31:0]             rdata;
     116    //write respond channel
     117    input  logic                    bready;
     118    output logic                    bvalid;
     119    output logic [1:0]              bresp;
     120    output logic [7:0]              bid;
     121  //arbiter inputs
     122  input logic rd_trans_done_i;
     123  input logic wr_trans_done_i;
     124  input logic [31:0] prdata_i;
     125  input logic err_of_transfer_i;
     126  input logic write_to_rd_sfifo_i;
     127  input logic read_from_wd_sfifo_i;
     128  input logic dec_error_i;
     129  input logic latch_resp_i;
     130  	//req
     131  output logic sfifo_aw_empty_o;
     132  output logic sfifo_ar_empty_o;
     133  	//data
     134  output logic sfifo_wd_almost_empty_o;
     135  output logic sfifo_rd_almost_full_o;
     136  //	
     137  output logic [31:0] write_burst_addr_o;
     138  output logic [7:0]  write_burst_len_o;
     139  output logic [2:0]  write_burst_size_o;
     140  output logic [1:0]  write_burst_name_o;
     141  output logic [2:0]  write_burst_prot_o;
     142  //
     143  output logic [31:0] read_burst_addr_o;
     144  output logic [7:0]  read_burst_len_o;
     145  output logic [2:0]  read_burst_size_o;
     146  output logic [1:0]  read_burst_name_o;
     147  output logic [2:0]  read_burst_prot_o;
     148  //
     149  output logic [31:0] wdata_to_apb_o;
     150  output logic [3:0] wstrb_to_apb_o;
     151    //******************************************************
     152    //internal signals
     153    //******************************************************
     154    //SFIFO_AW
     155    logic                           sfifo_aw_full;
     156    logic                           sfifo_aw_we;
     157    logic                           sfifo_aw_re;
     158    logic [7:0]                     sfifo_aw_id;
     159    //SFIFO_AR
     160    logic                           sfifo_ar_full;
     161    logic                           sfifo_ar_we;
     162    logic                           sfifo_ar_re;
     163    logic [7:0]                     sfifo_ar_id;
     164    //SFIFO_WD
     165    logic                           sfifo_wd_full;
     166    logic                           sfifo_wd_we;
     167    logic                           sfifo_wd_re;
     168    logic [31:0]                    sfifo_wd_wdata;
     169    logic [3:0]                     sfifo_wd_wstrb;
     170    logic 			  sfifo_wd_last;
     171    logic	 			  wlast_flag;
     172    //SFIFO_RD
     173    logic                           sfifo_rd_empty;
     174    logic                           sfifo_rd_we;
     175    logic                           sfifo_rd_re;
     176    //RD_CH
     177    logic[1:0]                      resp_of_rdata;
     178    logic 			  rdata_last;
     179    //WRITE RESPONSE
     180    logic bresp_reg;
     181    //
     182    parameter logic [POINTER_WIDTH:0] ALMOST_FULL_VALUE = 2**POINTER_WIDTH - 1'b1;
     183    parameter logic [POINTER_WIDTH:0] ALMOST_EMPTY_VALUE = 2**POINTER_WIDTH - 4'd14;
     184  //*******************************************************************
     185  //X2P_SFIFO_AR
     186  //*******************************************************************
     187    sfiforeq #(
     188  	.DATA_WIDTH(X2P_SFIFO_AR_DATA_WIDTH),
     189         	.POINTER_WIDTH(POINTER_WIDTH)
     190    ) ar_sfifo (
     191    .clk(aclk),
     192    .rst_n(aresetn),
     193    .wr(sfifo_ar_we),
     194    .rd(sfifo_ar_re),
     195    .data_in({araddr[31:0], arid[7:0], arlen[7:0], arsize[2:0], arburst[1:0], arprot[2:0]}),
     196    .sfifo_empty(sfifo_ar_empty_o),
     197    .sfifo_full(sfifo_ar_full),
     198    .data_out({read_burst_addr_o[31:0], sfifo_ar_id[7:0], read_burst_len_o[7:0], 
     199  	  read_burst_size_o[2:0], read_burst_name_o[1:0], read_burst_prot_o[2:0]})
     200    //
     201    //
     202    );
     203    //Logic
     204    assign arready  	= ~sfifo_ar_full;
     205    assign sfifo_ar_we 	= arready & arvalid;
     206    assign sfifo_ar_re 	= rd_trans_done_i;
     207    //*******************************************************************
     208    //X2P_SFIFO_RD
     209    //*******************************************************************
     210    sfifordata #(
     211  	.DATA_WIDTH(X2P_SFIFO_RD_DATA_WIDTH), .POINTER_WIDTH(POINTER_WIDTH),
     212  	.ALMOST_FULL_VALUE(ALMOST_FULL_VALUE), .ALMOST_EMPTY_VALUE(ALMOST_EMPTY_VALUE)
     213  ) rd_sfifo(
     214    .clk(aclk),
     215    .rst_n(aresetn),
     216    .wr(sfifo_rd_we),
     217    .rd(sfifo_rd_re),
     218    .data_in({rd_trans_done_i, resp_of_rdata[1:0], sfifo_ar_id[7:0], prdata_i[31:0]}),
     219    .sfifo_empty(sfifo_rd_empty),
     220    .sfifo_almost_full(sfifo_rd_almost_full_o),
     221    .data_out({rdata_last, rresp[1:0], rid[7:0], rdata[31:0]})
     222    );
     223    //Logic
     224    assign rvalid      = ~sfifo_rd_empty;
     225    assign sfifo_rd_re = rvalid & rready;
     226    assign sfifo_rd_we = write_to_rd_sfifo_i;
     227    //RD_CH
     228    //resp_of_rdata
     229    always_comb begin
     230      if(~err_of_transfer_i)
     231  	  resp_of_rdata = OKAY;
     232  	else if(dec_error_i)
     233  	  resp_of_rdata = DECERR;
     234  	else
     235  	  resp_of_rdata = PSLVERR;
     236    end
     237    //rlast
     238    always_comb begin
     239  	if(rvalid & rdata_last)
     240  	  rlast = 1'b1;
     241  	else
     242  	  rlast = 1'b0;	
     243    end
     244    //*******************************************************************
     245    //X2P_SFIFO_AW
     246    //*******************************************************************
     247    sfiforeq #(
     248  	  .DATA_WIDTH(X2P_SFIFO_AW_DATA_WIDTH),
     249  	  .POINTER_WIDTH(POINTER_WIDTH)
     250  ) aw_sfifo(
     251    .clk(aclk),
     252    .rst_n(aresetn),
     253    .wr(sfifo_aw_we),
     254    .rd(sfifo_aw_re),
     255    .data_in({awaddr[31:0], awid[7:0], awlen[7:0], awsize[2:0], awburst[1:0], awprot[2:0]}),
     256    .sfifo_empty(sfifo_aw_empty_o),
     257    .sfifo_full(sfifo_aw_full),
     258    .data_out({write_burst_addr_o[31:0], sfifo_aw_id[7:0], write_burst_len_o[7:0], 
     259  	  write_burst_size_o[2:0], write_burst_name_o[1:0], write_burst_prot_o[2:0]})
     260    );
     261    //Logic
     262    assign awready 	= ~sfifo_aw_full;
     263    assign sfifo_aw_we    = awready & awvalid;
     264    assign sfifo_aw_re    = wr_trans_done_i;
     265    //*******************************************************************
     266    //X2P_SFIFO_WD
     267    //*******************************************************************
     268    sfifowdata #(
     269  	.DATA_WIDTH(X2P_SFIFO_WD_DATA_WIDTH), .POINTER_WIDTH(POINTER_WIDTH),
     270  	.ALMOST_FULL_VALUE(ALMOST_FULL_VALUE), .ALMOST_EMPTY_VALUE(ALMOST_EMPTY_VALUE)
     271    ) wd_sfifo(
     272    .clk(aclk),
     273    .rst_n(aresetn),
     274    .wr(sfifo_wd_we),
     275    .rd(sfifo_wd_re),
     276    .data_in({wlast, wstrb[3:0], wdata[31:0]}),
     277    .sfifo_almost_empty(sfifo_wd_almost_empty_o),
     278    .sfifo_full(sfifo_wd_full),
     279    .data_out({wlast_flag, wstrb_to_apb_o[3:0], wdata_to_apb_o[31:0]})
     280    );
     281  //logic
     282  assign wready      = ~sfifo_wd_full;
     283  assign sfifo_wd_we = wvalid & wready;
     284  assign sfifo_wd_re = read_from_wd_sfifo_i;
     285  //
     286  //B CHANNEL
     287  //
     288  //bresp register
     289  //
     290  always_ff@(posedge aclk, negedge aresetn) begin
     291    if(~aresetn) bresp_reg <= 1'b0;
     292    else if(wr_trans_done_i) bresp_reg <= 1'b0; 
     293    else if(latch_resp_i) bresp_reg <= (bresp_reg | err_of_transfer_i);
     294  end
     295  //
     296  //INFO
     297  //
     298  always_ff @(posedge aclk, negedge aresetn) begin
     299  	if(~aresetn) begin
     300  	  bresp[1:0] <= OKAY;
     301  	  bid[7:0] <= 8'd0;
     302  	  bvalid <= 1'b0;
     303    	end
     304  	else if(wr_trans_done_i) begin
     305  		  bid[7:0] <= sfifo_aw_id[7:0];
     306  		  bvalid <= 1'b1;
     307  		  //
     308  		  if(~wlast_flag) 
     309  		    bresp[1:0] <= PSLVERR;
     310  		  else if(~bresp_reg)
     311  		    bresp[1:0] <= OKAY;
     312  		  else if(dec_error_i)
     313  		    bresp[1:0] <= DECERR;
     314  		  else
     315  		    bresp[1:0] <= PSLVERR;
     316  	end
     317  	else if(bready) begin
     318  	  bresp[1:0] <= OKAY;
     319  	  bid[7:0] <= 8'd0;
     320  	  bvalid <= 1'b0;
     321  	end
     322  end
     323  //
     324  //
     325  //
     326  endmodule: axi_transaction_controller
