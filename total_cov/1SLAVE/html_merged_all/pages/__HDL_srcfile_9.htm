     3  //
     4    //overall operation mechanism
     5    //(R1) granting read --write waits -> grant write for next time
     6    //(R2) granting read --ONLY read waits OR no request -> ... read ...
     7    //(W1) granting write --read waits -> ... read ...
     8    //(W2) granting write--ONLY write waits OR no request -> ... write ...
     9  module arbiter(
     10  	aclk,
     11  	aresetn,
     12  	pclk,
     13  	preset_n,
     14  	sfifo_ar_empty_i,
     15  	sfifo_aw_empty_i,
     16  	sfifo_rd_almost_full_i,
     17  	sfifo_wd_almost_empty_i,
     18  	bchannel_rdy_i,
     19  	//AXI REQ INFO
     20  	//WRITE
     21  	  write_burst_addr_i,
     22  	  write_burst_len_i,
     23  	  write_burst_size_i,
     24  	  write_burst_name_i,
     25  	  write_burst_prot_i,
     26  	//READ	
     27  	  read_burst_addr_i,
     28  	  read_burst_len_i,
     29  	  read_burst_size_i,
     30  	  read_burst_name_i,
     31  	  read_burst_prot_i,
     32  	//apb master input
     33  	addr_incr_en_i,
     34  	//counter input
     35  	burst_almost_done_i,
     36  	burst_done_i,
     37  	//
     38  	transfer_addr_o,
     39  	selected_addr_o,
     40  	selected_len_o,
     41  	selected_prot_o,
     42  	next_transfer_rdy_o,
     43  	disallowed_trans_o,
     44  	final_brsp_vld_o,
     45  	wr_trans_done_o,
     46  	rd_trans_done_o,
     47  	write_enable_o
     48  );
     49  
     50  //iclude parameter file
     51  import parameter_pkg::*; 
     52  //*****************************************************************
     53  //ports
     54  //*****************************************************************
     55  input logic aclk;
     56  input logic aresetn;
     57  input logic pclk;
     58  input logic preset_n;
     59  //req
     60  input logic sfifo_ar_empty_i;
     61  input logic sfifo_aw_empty_i;
     62  //data
     63  input logic sfifo_rd_almost_full_i;
     64  input logic sfifo_wd_almost_empty_i;
     65  input logic bchannel_rdy_i;
     66  	//AXI REQ INFO
     67  	//WRITE
     68  input logic [31:0] write_burst_addr_i;
     69  input logic [7:0] write_burst_len_i;
     70  input logic [2:0] write_burst_size_i;
     71  input logic [1:0] write_burst_name_i;
     72  input logic [2:0] write_burst_prot_i;
     73  	//READ	
     74  input logic [31:0] read_burst_addr_i;
     75  input logic [7:0] read_burst_len_i;
     76  input logic [2:0] read_burst_size_i;
     77  input logic [1:0] read_burst_name_i;
     78  input logic [2:0] read_burst_prot_i;
     79  //apb master input
     80  input logic addr_incr_en_i;
     81  //counter input
     82  input logic burst_almost_done_i;
     83  input logic burst_done_i;
     84  //
     85  output logic [31:0] transfer_addr_o;
     86  output logic [31:0] selected_addr_o;
     87  output logic [7:0] selected_len_o;
     88  output logic [2:0] selected_prot_o;
     89  output logic next_transfer_rdy_o;
     90  output logic disallowed_trans_o;
     91  output logic final_brsp_vld_o;
     92  output logic wr_trans_done_o;
     93  output logic rd_trans_done_o;
     94  output logic write_enable_o;
     95  //*****************************************************************
     96  //variables
     97  //*****************************************************************
     98  //ABT state machine
     99  typedef enum logic [1:0] {ABT_IDLE, ABT_GO, ABT_DONE, BRSP_FIFO_FULL} abt_st;
     100    abt_st abt_cs, abt_ns;
     101    //
     102    logic [1:0] nextSel;
     103    logic [1:0] nextGrant;
     104    logic [1:0] now_grant;
     105    logic rd_req_avail, wr_req_avail;
     106    logic update;
     107    logic begin_transfer;
     108    logic burst_go;
     109    logic burst_en;
     110    logic new_req_rdy;
     111    logic [1:0] control;
     112    logic [1:0] selected_burst;
     113    logic [2:0] selected_size;
     114    //addr variables
     115    logic addr_incr_active;
     116    logic [31:0] addr_reg;
     117    logic [31:0] next_addr_for_incr;
     118    logic [31:0] next_addr_for_wrap;
     119    logic [2:0] bit_num;
     120    logic [2:0] bit3Addr;
     121    logic [3:0] bit4Addr;
     122    logic [4:0] bit5Addr;
     123    logic [5:0] bit6Addr;
     124    logic invalid_transfer;
     125    //***************************************************
     126    //output assignment
     127    //***************************************************
     128  assign transfer_addr_o = addr_reg;
     129  assign selected_prot_o = now_grant[0] ? read_burst_prot_i : write_burst_prot_i;
     130  assign selected_addr_o = now_grant[0] ? read_burst_addr_i : write_burst_addr_i;
     131  assign selected_len_o = now_grant[0] ? read_burst_len_i : write_burst_len_i;
     132  assign wr_trans_done_o = now_grant[1] & control[1];// 
     133  assign rd_trans_done_o = now_grant[0] & control[1];
     134  assign write_enable_o = now_grant[1];
     135  //***************************************************
     136  //internal assignment
     137  //***************************************************
     138  assign selected_size = now_grant[0] ? read_burst_size_i : write_burst_size_i;
     139  assign selected_burst = now_grant[0] ? read_burst_name_i : write_burst_name_i;
     140  assign rd_req_avail = ~sfifo_ar_empty_i;
     141  assign wr_req_avail = ~sfifo_aw_empty_i;
     142  assign burst_en = new_req_rdy ? (now_grant[0] ? ~sfifo_ar_empty_i : ~sfifo_aw_empty_i) : 1'b0;
     143  assign invalid_transfer = ((selected_burst != 2'b01) && (|selected_addr_o[1:0])) | (selected_size != 3'b010); 
     144  //
     145  //ready to run new request
     146  //
     147    always_ff@(posedge aclk, negedge aresetn) begin
     148      if(~aresetn) new_req_rdy <= 1'b1;
     149      else if(control[0]) begin
     150        new_req_rdy <= 1'b0;
     151      end
     152      else if(control[1]) begin
     153        new_req_rdy <= 1'b1;
     154      end 
     155    end
     156  //
     157  //state reg
     158  //
     159  always_ff@(posedge aclk, negedge aresetn) begin
     160  	if(~aresetn) abt_cs <= ABT_IDLE;
     161  	else abt_cs <= abt_ns;
     162  end
     163  //
     164  always_comb begin
     165    abt_ns = ABT_IDLE;
     166    burst_go = 1'b0;
     167    begin_transfer = 1'b0;
     168    control = 2'b0;
     169    next_transfer_rdy_o = 1'b0;
     170    addr_incr_active = 1'b0;
     171    final_brsp_vld_o = 1'b0;
     172    //
     173    case(abt_cs)
     174    ABT_IDLE: begin
     175  	  if(burst_en) begin
     176  		  abt_ns = ABT_GO;
     177  		  begin_transfer = 1'b1;
     178  		  control = 2'b01;
     179  	  end
     180  	  else begin
     181  		  abt_ns = ABT_IDLE;
     182  		  control = 2'b00;
     183  	  end
     184    end
     185    ABT_GO: begin
     186  	if(burst_almost_done_i) abt_ns = ABT_DONE;//SET_UP phase
     187  	else abt_ns = ABT_GO;
     188  	//
     189  	burst_go = 1'b1;
     190  	addr_incr_active = addr_incr_en_i;
     191    	next_transfer_rdy_o = now_grant[0] ? ~sfifo_rd_almost_full_i : ~sfifo_wd_almost_empty_i;
     192    end
     193    ABT_DONE: begin
     194  	  next_transfer_rdy_o = 1'b0;
     195  	  if(burst_done_i) begin //rd AW_FIFO or AR_FIFO
     196  		  if(now_grant[0]) begin
     197  			  abt_ns = ABT_IDLE;
     198  			  control = 2'b10;
     199  		  end
     200  		  else begin
     201  		  final_brsp_vld_o = 1'b1;
     202  		  //
     203  			  if(bchannel_rdy_i) begin
     204  				  control = 2'b10;
     205  				  abt_ns = ABT_IDLE;
     206  			  end
     207  			  else begin
     208  				  control = 2'b00;
     209  				  abt_ns = BRSP_FIFO_FULL;
     210  			  end
     211  		  end
     212  	  end
     213  	  else abt_ns = ABT_DONE;
     214    end
     215    BRSP_FIFO_FULL: begin
     216  	  next_transfer_rdy_o = 1'b0;
     217  	  //
     218  	  if(bchannel_rdy_i) begin
     219  		  control = 2'b10;
     220  		  abt_ns = ABT_IDLE;
     221  	  end
     222  	  else begin
     223  		  control = 2'b00;
     224  		  abt_ns = BRSP_FIFO_FULL;
     225  	  end
     226    end
     227    endcase
     228  end
     229  
     230    //
     231    //nextSel0
     232    always_comb begin
     233      if(now_grant[0])
     234  	  nextSel[0] = 1'b1;
     235  	else if(rd_req_avail)
     236  	  nextSel[0] = 1'b0;
     237  	else
     238  	  nextSel[0] = 1'b1;
     239    end
     240    //nextSel1
     241    always_comb begin
     242      if(now_grant[1])
     243  	  nextSel[1] = 1'b1;
     244  	else if(wr_req_avail)
     245  	  nextSel[1] = 1'b0;
     246  	else
     247  	  nextSel[1] = 1'b1;
     248    end
     249    //nextGrant[1]
     250    always_comb begin
     251      if(~nextSel[0])
     252  	  nextGrant[1] = 1'b0;
     253  	else if(wr_req_avail) //	R1
     254  	  nextGrant[1] = 1'b1;
     255  	else
     256  	  nextGrant[1] = now_grant[1]; //R2
     257    end
     258    //nextGrant[0]
     259    always_comb begin
     260      if(~nextSel[1])
     261  	  nextGrant[0] = 1'b0;
     262  	else if(rd_req_avail) //W1
     263  	  nextGrant[0] = 1'b1;
     264  	else
     265  	  nextGrant[0] = now_grant[0]; //W2
     266    end
     267  //
     268  //now_grant
     269  //
     270  assign update = (burst_go ? 1'b0 
     271  : (now_grant[0] ? (wr_req_avail & sfifo_ar_empty_i) : (rd_req_avail & sfifo_aw_empty_i))) | control[1];
     272  //
     273  always_ff @(posedge aclk, negedge aresetn) begin
     274  	if(~aresetn)
     275  		now_grant[1:0] <= 2'b01;
     276  	else if(update)
     277  		now_grant[1:0] <= nextGrant[1:0];
     278  	else
     279  		now_grant[1:0] <= now_grant[1:0];	  
     280  end
     281  //
     282  //address register
     283  //
     284  always_ff@(posedge pclk, negedge preset_n) begin
     285  	if(~preset_n) begin
     286  		addr_reg <= 32'd0;
     287  		disallowed_trans_o <= 1'b0;
     288  	end
     289  	else if(begin_transfer) begin
     290  		addr_reg <= selected_addr_o[31:0];//, 2'b00};
     291  		disallowed_trans_o <= invalid_transfer;
     292  	end
     293  	else if(addr_incr_active) begin
     294  		case(selected_burst) 
     295  			2'b00: addr_reg <= addr_reg;
     296  			2'b01: addr_reg <= {next_addr_for_incr[31:2], 2'b00};
     297  			2'b10: addr_reg <= next_addr_for_wrap;
     298  			default: addr_reg <= 32'd0;
     299  		endcase
     300  	end
     301  end
     302  //
     303  //calculate address
     304  //
     305  assign next_addr_for_incr[31:0] = addr_reg[31:0] + 3'd4;
     306  //
     307  //bit_num
     308  //
     309  always_comb begin
     310    bit_num[2:0] = 3'b000;
     311  case(selected_len_o[7:0])
     312    8'd1:  bit_num[2:0] = 3'b011;
     313    8'd3:  bit_num[2:0] = 3'b100;
     314    8'd7:  bit_num[2:0] = 3'b101;
     315    8'd15: bit_num[2:0] = 3'b110;
     316    default: bit_num[2:0] = 3'b000;
     317  endcase
     318  end
     319  //bit3Addr, bit4Addr, bit5Addr, bit6Addr
     320  always_comb begin
     321    bit3Addr[2:0] = 3'd0;
     322    //
     323  if(bit_num[2:0] == 3'b011)
     324    bit3Addr[2:0] = addr_reg[2:0] + 3'd4;
     325  else
     326    bit3Addr[2:0] = 3'd0;
     327  end
     328  always_comb begin
     329    bit4Addr[3:0] = 4'd0;
     330    //
     331  if(bit_num[2:0] == 3'b100)
     332    bit4Addr[3:0] = addr_reg[3:0] + 3'd4;
     333  else
     334    bit4Addr[3:0] = 4'd0;
     335  end
     336  always_comb begin
     337    bit5Addr[4:0] = 5'd0;
     338    //
     339  if(bit_num[2:0] == 3'b101)
     340    bit5Addr[4:0] = addr_reg[4:0] + 3'd4;
     341  else
     342    bit5Addr[4:0] = 5'd0;
     343  end
     344  always_comb begin
     345    bit6Addr[5:0] = 6'd0;
     346    //
     347  if(bit_num[2:0] == 3'b110)
     348    bit6Addr[5:0] = addr_reg[5:0] + 3'd4;
     349  else
     350    bit6Addr[5:0] = 6'd0;
     351  end
     352  //next_addr_for_wrap
     353  always_comb begin
     354    next_addr_for_wrap[31:0] = 32'd0;
     355  case(bit_num[2:0])
     356    3'b011: next_addr_for_wrap[31:0] = {addr_reg[31:3], bit3Addr[2:0]};
     357    3'b100: next_addr_for_wrap[31:0] = {addr_reg[31:4], bit4Addr[3:0]};
     358    3'b101: next_addr_for_wrap[31:0] = {addr_reg[31:5], bit5Addr[4:0]};
     359    3'b110: next_addr_for_wrap[31:0] = {addr_reg[31:6], bit6Addr[5:0]};
     360    default: next_addr_for_wrap[31:0] = 32'd0;
     361  endcase
     362  end
     363  endmodule
